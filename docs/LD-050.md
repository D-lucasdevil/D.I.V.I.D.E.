    /* ══════════════════════════════════
       FIRE CANVAS — flame simulation
    ══════════════════════════════════ */
    (function() {
      const canvas = document.getElementById('fireCanvas');
      const ctx = canvas.getContext('2d');
      let W, H, cols, buf;
      const CELL = 4;
      const COOLING = 55;

      function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
        cols = Math.ceil(W / CELL);
        const rows = Math.ceil(H / CELL) + 2;
        buf = new Uint8Array(cols * rows).fill(0);
      }

      function heatColor(t) {
        // black → red → orange → yellow → white
        const r = Math.min(255, t * 3);
        const g = Math.max(0, Math.min(255, t * 3 - 255));
        const b = Math.max(0, Math.min(255, t * 3 - 510));
        return `rgb(${r|0},${g|0},${b|0})`;
      }

      function step() {
        const rows = Math.ceil(H / CELL) + 2;
        // seed bottom row
        for (let x = 0; x < cols; x++) {
          buf[(rows - 1) * cols + x] = Math.random() > 0.4
            ? 200 + Math.random() * 55 | 0
            : 0;
        }
        // propagate upward — only at edges (left ~15% and right ~15%)
        for (let y = rows - 2; y >= 0; y--) {
          for (let x = 0; x < cols; x++) {
            const edgeFraction = x / cols;
            const inEdge = edgeFraction < 0.15 || edgeFraction > 0.85;
            if (!inEdge) { buf[y * cols + x] = 0; continue; }
            const b0 = buf[(y+1)*cols + x] || 0;
            const b1 = buf[(y+1)*cols + Math.max(0,x-1)] || 0;
            const b2 = buf[(y+1)*cols + Math.min(cols-1,x+1)] || 0;
            const b3 = buf[(y+1)*cols + x] || 0;
            const avg = ((b0 + b1 + b2 + b3) / 4) - COOLING;
            buf[y * cols + x] = Math.max(0, avg + (Math.random() * 10 - 5)) | 0;
          }
        }
        ctx.clearRect(0, 0, W, H);
        for (let y = 0; y < rows - 1; y++) {
          for (let x = 0; x < cols; x++) {
            const t = buf[y * cols + x];
            if (t < 5) continue;
            ctx.fillStyle = heatColor(t);
            ctx.fillRect(x * CELL, y * CELL, CELL, CELL);
          }
        }
        requestAnimationFrame(step);
      }

      resize();
      window.addEventListener('resize', resize);
      step();
    })();

    /* ══════════════════════════════════
       BURN REVEAL — canvas fire wipe

    let burnStarted = false;

    function startBurnReveal() {
      if (burnStarted) return;
      burnStarted = true;

      const seal = document.getElementById('imageSeal');
      const container = document.getElementById('imageContainer');
      const burnCanvas = document.getElementById('burnCanvas');
      const caption = document.getElementById('imageCaption');
      const ctx = burnCanvas.getContext('2d');

      const W = container.offsetWidth;
      const H = container.offsetHeight;
      burnCanvas.width = W;
      burnCanvas.height = H;

      // Draw the seal's content onto the burn canvas as a solid fill first
      ctx.fillStyle = '#0d0100';
      ctx.fillRect(0, 0, W, H);

      // Hide the HTML seal immediately — canvas takes over
      seal.style.display = 'none';

      // Burn cells
      const CELL = 6;
      const cols = Math.ceil(W / CELL);
      const rows = Math.ceil(H / CELL);
      const total = cols * rows;
      // Each cell: 0=sealed, 1=burning, 2=gone
      const state = new Uint8Array(total).fill(0);
      const heat  = new Float32Array(total).fill(0);
      // Ignition front — start from bottom
      let ignited = 0;

      // Seed bottom row
      for (let x = 0; x < cols; x++) {
        const idx = (rows - 1) * cols + x;
        state[idx] = 1;
        heat[idx] = 1.0;
        ignited++;
      }

      function cellColor(h, s) {
        if (s === 2) return null; // transparent
        if (s === 1) {
          // burning — orange/yellow
          const r = 255;
          const g = Math.min(255, h * 300) | 0;
          const b = Math.min(255, h * 50) | 0;
          return `rgba(${r},${g},${b},${(0.5 + h * 0.5).toFixed(2)})`;
        }
        return '#0d0100'; // sealed
      }

      let frame = 0;
      function burnFrame() {
        frame++;
        // Spread burn upward + sideways
        const newState = new Uint8Array(state);
        const newHeat = new Float32Array(heat);

        for (let r = rows - 1; r >= 0; r--) {
          for (let c = 0; c < cols; c++) {
            const idx = r * cols + c;
            if (state[idx] === 1) {
              newHeat[idx] = Math.max(0, heat[idx] - 0.04 - Math.random() * 0.04);
              if (newHeat[idx] <= 0) {
                newState[idx] = 2; // burned through
                ignited--;
              }
              // Spread up
              if (r > 0 && Math.random() > 0.35) {
                const up = (r-1) * cols + c;
                if (state[up] === 0) { newState[up] = 1; newHeat[up] = 0.8 + Math.random() * 0.2; ignited++; }
              }
              // Spread left/right
              if (c > 0 && Math.random() > 0.6) {
                const left = r * cols + c - 1;
                if (state[left] === 0) { newState[left] = 1; newHeat[left] = 0.6 + Math.random() * 0.3; ignited++; }
              }
              if (c < cols-1 && Math.random() > 0.6) {
                const right = r * cols + c + 1;
                if (state[right] === 0) { newState[right] = 1; newHeat[right] = 0.6 + Math.random() * 0.3; ignited++; }
              }
            }
          }
        }

        state.set(newState);
        heat.set(newHeat);

        // Draw
        ctx.clearRect(0, 0, W, H);
        for (let r = 0; r < rows; r++) {
          for (let c = 0; c < cols; c++) {
            const idx = r * cols + c;
            const s = state[idx];
            if (s === 2) continue;
            const col = cellColor(heat[idx], s);
            if (col) {
              ctx.fillStyle = col;
              ctx.fillRect(c * CELL, r * CELL, CELL, CELL);
            }
          }
        }

        // Add fire glow on burn front
        for (let r = 0; r < rows; r++) {
          for (let c = 0; c < cols; c++) {
            const idx = r * cols + c;
            if (state[idx] === 1 && heat[idx] > 0.5) {
              ctx.fillStyle = `rgba(255,${(heat[idx]*200)|0},0,0.6)`;
              ctx.fillRect(c * CELL, r * CELL, CELL, CELL);
            }
          }
        }

        // Check if all burned
        const allGone = state.every(s => s === 2);
        if (!allGone) {
          requestAnimationFrame(burnFrame);
        } else {
          // Done — hide canvas, show caption
          burnCanvas.style.display = 'none';
          caption.classList.add('show');
        }
      }

      requestAnimationFrame(burnFrame);
    }

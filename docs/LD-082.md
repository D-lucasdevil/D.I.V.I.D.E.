    /* ══════════════════════════════════
       MIST CANVAS SIMULATION

    (function() {
      const canvas = document.getElementById('mistCanvas');
      const ctx = canvas.getContext('2d');
      let W, H;
      const blobs = [];

      function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
      }

      function makeBlob() {
        return {
          x: Math.random() * W,
          y: Math.random() * H,
          r: 80 + Math.random() * 200,
          vx: (Math.random() - 0.5) * 0.3,
          vy: (Math.random() - 0.5) * 0.15,
          alpha: 0.02 + Math.random() * 0.06,
          phase: Math.random() * Math.PI * 2,
          speed: 0.003 + Math.random() * 0.004
        };
      }

      for (let i = 0; i < 30; i++) blobs.push(makeBlob());

      function draw() {
        ctx.clearRect(0, 0, W, H);
        const t = performance.now() / 1000;
        blobs.forEach(b => {
          b.x += b.vx;
          b.y += b.vy;
          if (b.x < -b.r * 2) b.x = W + b.r;
          if (b.x > W + b.r * 2) b.x = -b.r;
          if (b.y < -b.r * 2) b.y = H + b.r;
          if (b.y > H + b.r * 2) b.y = -b.r;
          const a = b.alpha * (0.6 + 0.4 * Math.sin(t * b.speed + b.phase));
          const grad = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.r);
          grad.addColorStop(0, `rgba(100,160,190,${a})`);
          grad.addColorStop(0.5, `rgba(80,130,160,${a * 0.5})`);
          grad.addColorStop(1, 'rgba(60,100,130,0)');
          ctx.fillStyle = grad;
          ctx.beginPath();
          ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
          ctx.fill();
        });
        requestAnimationFrame(draw);
      }

      resize();
      window.addEventListener('resize', resize);
      draw();
    })();

    /* ══════════════════════════════════
       FOG PARTING REVEAL
    ══════════════════════════════════ */
    let fogStarted = false;

    function startFogReveal() {
      if (fogStarted) return;
      fogStarted = true;

      const seal = document.getElementById('imageSeal');
      const container = document.getElementById('imageContainer');
      const fogCanvas = document.getElementById('fogCanvas');
      const caption = document.getElementById('imageCaption');
      const ctx = fogCanvas.getContext('2d');

      const W = container.offsetWidth;
      const H = container.offsetHeight;
      fogCanvas.width = W;
      fogCanvas.height = H;

      // Draw initial fog cover
      ctx.fillStyle = '#060e14';
      ctx.fillRect(0, 0, W, H);
      seal.style.display = 'none';

      // Fog particles — each is a fog blob that slowly dissipates from center outward
      const BLOBS = [];
      const COUNT = 120;
      const cx = W / 2, cy = H / 2;

      for (let i = 0; i < COUNT; i++) {
        const angle = Math.random() * Math.PI * 2;
        const dist = Math.random() * Math.max(W, H) * 0.6;
        BLOBS.push({
          x: cx + Math.cos(angle) * dist,
          y: cy + Math.sin(angle) * dist,
          r: 40 + Math.random() * 80,
          alpha: 0.8 + Math.random() * 0.2,
          vx: (Math.random() - 0.5) * 1.5,
          vy: (Math.random() - 0.5) * 0.8,
          decay: 0.004 + Math.random() * 0.006,
          drift: Math.random() * 0.02
        });
      }

      // Center clears first — add outward radial velocity
      BLOBS.forEach(b => {
        const dx = b.x - cx, dy = b.y - cy;
        const dist = Math.sqrt(dx*dx + dy*dy) || 1;
        b.vx += (dx / dist) * 0.8;
        b.vy += (dy / dist) * 0.4;
      });

      let frame = 0;
      function fogFrame() {
        frame++;
        ctx.clearRect(0, 0, W, H);

        // Dark background
        ctx.fillStyle = '#060e14';
        ctx.fillRect(0, 0, W, H);

        let anyVisible = false;

        BLOBS.forEach(b => {
          b.x += b.vx;
          b.y += b.vy;
          b.vx += (Math.random() - 0.5) * 0.1;
          b.vy += (Math.random() - 0.5) * 0.05;
          b.alpha -= b.decay;
          if (b.alpha <= 0) return;
          anyVisible = true;

          const a = Math.max(0, b.alpha);
          const grad = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.r);
          grad.addColorStop(0, `rgba(80,120,150,${a})`);
          grad.addColorStop(0.5, `rgba(60,100,130,${a * 0.6})`);
          grad.addColorStop(1, 'rgba(40,70,90,0)');
          ctx.fillStyle = grad;
          ctx.beginPath();
          ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
          ctx.fill();
        });

        if (anyVisible) {
          requestAnimationFrame(fogFrame);
        } else {
          fogCanvas.style.display = 'none';
          caption.classList.add('show');
        }
      }

      requestAnimationFrame(fogFrame);
    }

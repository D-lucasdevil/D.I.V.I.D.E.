          shards.push({ a: dir + (Math.random() - 0.5) * 1.5, L: L, w: Math.max(10 * sc, rand(0.13, 0.28) * L), sh: rand(0.5, 0.72), skew: rand(-0.12, 0.12), off: rand(-1, 1) * 26 * sc, ph: rand(0, TAU) });
        }
        return { x: x, y: y, dir: dir, thr: thr, shards: shards };
      }

      function layout() {
        backC = []; frontC = []; var i;
        /* back layer: large, dim, behind content */
        for (i = 0; i < 26; i++) backC.push(mkCluster(rand(0, W), H + 10, -Math.PI / 2, rand(0.03, 0.8), true));
        for (i = 0; i < 9; i++) backC.push(mkCluster(-10, rand(0.2, 1) * H, 0, rand(0.1, 0.8), true));
        for (i = 0; i < 9; i++) backC.push(mkCluster(W + 10, rand(0.2, 1) * H, Math.PI, rand(0.1, 0.8), true));
        /* front layer: hugging edges, over the content */
        var nb = Math.max(14, Math.floor(W / 52));
        for (i = 0; i < nb; i++) {
          var x = rand(0, W), edgeX = (x < W * 0.18 || x > W * 0.82), c = mkCluster(x, H + 6, -Math.PI / 2, edgeX ? rand(0.02, 0.6) : rand(0.5, 0.97), false);
          if (!edgeX) c.shards.forEach(function (s) { s.L *= 0.55; s.w *= 0.7; });
          frontC.push(c);
        }
        for (i = 0; i < 11; i++) frontC.push(mkCluster(-8, rand(0.05, 0.95) * H, 0, rand(0.05, 0.9), false));
        for (i = 0; i < 11; i++) frontC.push(mkCluster(W + 8, rand(0.05, 0.95) * H, Math.PI, rand(0.05, 0.9), false));
        for (i = 0; i < 12; i++) frontC.push(mkCluster(rand(0, W), -6, Math.PI / 2, rand(0.45, 0.95), false));
        motes = [];
        for (i = 0; i < 90; i++) motes.push({ x: rand(0, W), y: rand(0, H), vy: -rand(8, 30), vx: rand(-8, 8), s: rand(0.8, 2.6), ph: rand(0, TAU), a: rand(0.3, 0.8), need: Math.random() });
        drips = [];
      }

      function drawShard(ctx, bx, by, a, L, w, sh, skew, alpha, hot) {
        var ux = Math.cos(a), uy = Math.sin(a), vx = -uy, vy = ux, hw = w / 2;
        var p1x = bx - vx * hw, p1y = by - vy * hw, p2x = bx + vx * hw, p2y = by + vy * hw;
        var sx = bx + ux * L * sh, sy = by + uy * L * sh;
        var s1x = sx - vx * hw * 0.92, s1y = sy - vy * hw * 0.92, s2x = sx + vx * hw * 0.92, s2y = sy + vy * hw * 0.92;
        var tx = bx + ux * L + vx * w * skew, ty = by + uy * L + vy * w * skew;
        var g = ctx.createLinearGradient(bx, by, tx, ty);
        g.addColorStop(0, 'rgba(' + Math.round(50 + 30 * hot) + ',0,10,' + alpha + ')');
        g.addColorStop(0.55, 'rgba(' + Math.round(150 + 40 * hot) + ',8,30,' + alpha + ')');
        g.addColorStop(1, 'rgba(255,' + Math.round(70 + 70 * hot) + ',' + Math.round(90 + 60 * hot) + ',' + alpha + ')');
        ctx.fillStyle = g; ctx.beginPath(); ctx.moveTo(p1x, p1y); ctx.lineTo(s1x, s1y); ctx.lineTo(tx, ty); ctx.lineTo(s2x, s2y); ctx.lineTo(p2x, p2y); ctx.closePath(); ctx.fill();
        ctx.fillStyle = 'rgba(0,0,0,' + (0.3 * alpha) + ')'; ctx.beginPath(); ctx.moveTo(bx, by); ctx.lineTo(tx, ty); ctx.lineTo(s2x, s2y); ctx.lineTo(p2x, p2y); ctx.closePath(); ctx.fill();
        ctx.lineWidth = 1; ctx.strokeStyle = 'rgba(255,190,200,' + (0.5 * alpha) + ')'; ctx.beginPath(); ctx.moveTo(bx, by); ctx.lineTo(tx, ty); ctx.stroke();
        ctx.strokeStyle = 'rgba(255,70,95,' + (0.45 * alpha) + ')'; ctx.beginPath(); ctx.moveTo(p1x, p1y); ctx.lineTo(s1x, s1y); ctx.lineTo(tx, ty); ctx.lineTo(s2x, s2y); ctx.lineTo(p2x, p2y); ctx.stroke();
        return [tx, ty];
      }

      function drawClusters(ctx, list, t, alpha, front) {
        var hot = 0.5 + 0.5 * ds, pulseF = 1.4 + 4 * ds;
        for (var i = 0; i < list.length; i++) {
          if (light && (i & 1)) continue;
          var c = list[i], g = ease(clamp((ds - c.thr) / 0.2, 0, 1));
          if (g <= 0.005) continue;
          var vx = -Math.sin(c.dir), vy = Math.cos(c.dir);
          var tips = [];
          for (var k = 0; k < c.shards.length; k++) {
            var s = c.shards[k];
            var pul = 1 + 0.02 * Math.sin(t * pulseF + s.ph);
            var tp = drawShard(ctx, c.x + vx * s.off, c.y + vy * s.off, s.a, s.L * g * pul, s.w * (0.4 + 0.6 * g), s.sh, s.skew, alpha, hot);
            tips.push(tp);
          }
          ctx.save(); ctx.globalCompositeOperation = 'lighter';
          var pa = 0.18 + 0.2 * (0.5 + 0.5 * Math.sin(t * pulseF + c.shards[0].ph));
          for (k = 0; k < tips.length; k++) { var r = 22 + 26 * g; ctx.globalAlpha = pa * g * (front ? 1 : 0.7); ctx.drawImage(GL, tips[k][0] - r, tips[k][1] - r, r * 2, r * 2); }
          var br = (60 + 140 * g) * sc; ctx.globalAlpha = 0.2 * g * (0.6 + 0.4 * ds); ctx.drawImage(GL, c.x - br, c.y - br, br * 2, br * 2);
          ctx.restore();
          if (front && !light && g > 0.7 && Math.random() < 0.0035 * (0.3 + ds) && drips.length < 40) {
            var tp0 = tips[(Math.random() * tips.length) | 0]; if (tp0[1] > 0 && tp0[1] < H && tp0[0] > 0 && tp0[0] < W) drips.push({ x: tp0[0], y: tp0[1], v: rand(40, 120), l: rand(14, 42) });
          }
        }
      }

      function drawPool(ctx, t) {
        var level = H * (0.012 + 0.13 * Math.pow(ds, 1.2));
        if (level < 3) return;
        var top = H - level, g = ctx.createLinearGradient(0, top, 0, H);
        g.addColorStop(0, 'rgba(150,0,18,0.0)'); g.addColorStop(0.15, 'rgba(140,0,20,0.55)'); g.addColorStop(1, 'rgba(60,0,8,0.92)');
        ctx.fillStyle = g; ctx.beginPath(); ctx.moveTo(0, H);
        for (var x = 0; x <= W; x += 12) ctx.lineTo(x, top + Math.sin(x * 0.018 + t * 1.3) * 3.5 + Math.sin(x * 0.05 - t * 0.9) * 2);
        ctx.lineTo(W, H); ctx.closePath(); ctx.fill();
        ctx.strokeStyle = 'rgba(255,90,110,' + (0.25 + 0.2 * ds) + ')'; ctx.lineWidth = 1.5; ctx.beginPath();
        for (x = 0; x <= W; x += 12) { var yy = top + Math.sin(x * 0.018 + t * 1.3) * 3.5 + Math.sin(x * 0.05 - t * 0.9) * 2; if (x === 0) ctx.moveTo(x, yy); else ctx.lineTo(x, yy); }
        ctx.stroke();
      }

      function drawDrips(ctx, dt) {
        var level = H * (0.012 + 0.13 * Math.pow(ds, 1.2)), floor = H - level;
        for (var i = drips.length - 1; i >= 0; i--) {
          var p = drips[i]; p.y += p.v * dt;
          if (p.y > floor) { drips.splice(i, 1); continue; }
          var g = ctx.createLinearGradient(p.x, p.y - p.l, p.x, p.y);
          g.addColorStop(0, 'rgba(120,0,16,0)'); g.addColorStop(1, 'rgba(255,50,75,0.9)');
          ctx.strokeStyle = g; ctx.lineWidth = 2; ctx.lineCap = 'round'; ctx.beginPath(); ctx.moveTo(p.x, p.y - p.l); ctx.lineTo(p.x, p.y); ctx.stroke();
          ctx.fillStyle = 'rgba(255,60,85,0.95)'; ctx.beginPath(); ctx.arc(p.x, p.y, 2.2, 0, TAU); ctx.fill();
        }
      }

      function drawMotes(ctx, dt, t) {
        ctx.save(); ctx.globalCompositeOperation = 'lighter';
        for (var i = 0; i < motes.length; i++) {
          var m = motes[i]; if (m.need > ds * 1.1) continue;
          m.y += m.vy * dt; m.x += (m.vx + Math.sin(t * 0.9 + m.ph) * 10) * dt;
          if (m.y < -20) { m.y = H + 10; m.x = rand(0, W); }
          ctx.globalAlpha = m.a * (0.4 + 0.6 * Math.sin(t * 2 + m.ph) * 0.5 + 0.3) * clamp(ds * 1.6, 0, 1);
          var s = m.s * 7; ctx.drawImage(GL, m.x - s, m.y - s, s * 2, s * 2);
        }
        ctx.restore();
      }

      var last = performance.now(), crysEl = $('crysVal');
      function frame(now) {
        requestAnimationFrame(frame);
        var dt = Math.min(0.05, (now - last) / 1000); last = now;
        ds += (d - ds) * Math.min(1, dt * 4);
        if (Math.abs(ds - shown) > 0.002) {
          shown = ds; root.style.setProperty('--d', ds.toFixed(3)); root.style.setProperty('--rp', (5 - 3.4 * ds).toFixed(2) + 's');
          crysEl.textContent = Math.round(ds * 100) + '%';
        }
        var t = light ? 0 : now / 1000;
        bctx.clearRect(0, 0, W, H); fctx.clearRect(0, 0, W, H);
        drawClusters(bctx, backC, t, 0.55, false);
        drawPool(fctx, t);
        drawClusters(fctx, frontC, t, 0.88, true);
        if (!light) { drawDrips(fctx, dt); drawMotes(fctx, dt, t); }
        audioTick(now);
      }

      /* ═══════ AUDIO (opt-in resonance) ═══════ */
      var ac = null, master = null, oscs = [], sndOn = false, lastA = 0;
      function audioInit() {
        var AC = window.AudioContext || window.webkitAudioContext; if (!AC) return false;
        ac = new AC(); master = ac.createGain(); master.gain.value = 0; master.connect(ac.destination);
        [55, 110.5, 1760, 2349.3].forEach(function (f, i) {
          var o = ac.createOscillator(); o.type = 'sine'; o.frequency.value = f;
          var g = ac.createGain(); g.gain.value = i < 2 ? 0.5 : 0.1; o.connect(g); g.connect(master); o.start(); oscs.push({ o: o, f: f, hi: i >= 2 });
        });
        return true;
      }
      function audioTick(now) {
        if (!ac || now - lastA < 120) return; lastA = now;
        var tt = ac.currentTime;
        master.gain.setTargetAtTime(sndOn ? (0.006 + 0.04 * ds) : 0, tt, 0.35);
        oscs.forEach(function (x) { x.o.frequency.setTargetAtTime(x.f * (1 + ds * (x.hi ? 0.14 : 0.03)), tt, 0.4); });
      }
      var sndBtn = $('sndBtn');
      sndBtn.addEventListener('click', function () {
        try {
          if (!ac && !audioInit()) { sndBtn.textContent = '\u266B Resonance: n/a'; return; }
          if (ac.state === 'suspended') ac.resume();
          sndOn = !sndOn;
          sndBtn.innerHTML = '&#9835; Resonance: ' + (sndOn ? 'on' : 'off'); sndBtn.classList.toggle('active', sndOn);
        } catch (e) { sndBtn.textContent = '\u266B Resonance: n/a'; }
      });

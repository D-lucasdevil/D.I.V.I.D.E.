    let revealStarted = false;

    function startReveal() {
      if (revealStarted) return;
      revealStarted = true;

      const seal = document.getElementById('imageSeal');
      const container = document.getElementById('imageContainer');
      const canvas = document.getElementById('sealCanvas');
      const caption = document.getElementById('imageCaption');
      const ctx = canvas.getContext('2d');

      const W = container.offsetWidth;
      const H = container.offsetHeight;
      canvas.width = W;
      canvas.height = H;

      // White fill
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(0, 0, W, H);
      seal.style.display = 'none';

      // Feather-drift reveal — cells fade from white center outward
      const CELL = 6;
      const cols = Math.ceil(W / CELL);
      const rows = Math.ceil(H / CELL);
      const cx = W / 2, cy = H / 2;

      const cells = [];
      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          const px = c*CELL + CELL/2, py = r*CELL + CELL/2;
          const dx = px-cx, dy = py-cy;
          const dist = Math.sqrt(dx*dx + dy*dy);
          const angle = Math.atan2(dy, dx);
          // Feather-like drift — radial with gentle waviness
          const wave = Math.sin(angle * 6) * 15;
          cells.push({ c, r, priority: dist + wave + Math.random()*12, done: false });
        }
      }
      cells.sort((a, b) => a.priority - b.priority);

      let idx = 0;
      let frame = 0;
      const maxPriority = cells[cells.length-1].priority;

      function revealFrame() {
        frame++;
        const targetP = (frame / 80) * (maxPriority + 20);

        while (idx < cells.length && cells[idx].priority < targetP) {
          const cell = cells[idx];
          cell.done = true;
          // Soft gold-white flash
          const intensity = Math.max(0, 1 - cell.priority / maxPriority);
          ctx.fillStyle = `rgba(212,184,112,${0.5 * intensity + 0.1})`;
          ctx.fillRect(cell.c*CELL, cell.r*CELL, CELL, CELL);
          const ci = idx;
          setTimeout(() => {
            if (cells[ci]) ctx.clearRect(cells[ci].c*CELL, cells[ci].r*CELL, CELL, CELL);
          }, 50 + Math.random() * 120);
          idx++;
        }

        if (idx < cells.length) {
          requestAnimationFrame(revealFrame);
        } else {
          setTimeout(() => {
            canvas.style.transition = 'opacity 0.6s';
            canvas.style.opacity = '0';
            setTimeout(() => {
              canvas.style.display = 'none';
              caption.classList.add('show');
            }, 600);
          }, 200);
        }
      }

      requestAnimationFrame(revealFrame);
    }
  </script>

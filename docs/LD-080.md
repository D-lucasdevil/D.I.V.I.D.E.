    /* ══ CABLE CANVAS ══ */
    (function() {
      const canvas = document.getElementById('cableCanvas');
      const ctx = canvas.getContext('2d');
      let W, H;
      const cables = [];

      function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
      }

      function makeCable() {
        const side = Math.random() > 0.5 ? 'left' : 'right';
        const startX = side === 'left' ? 0 : W;
        const startY = Math.random() * H;
        const endX = Math.random() * W;
        const endY = Math.random() * H;
        return {
          x1: startX, y1: startY,
          x2: endX, y2: endY,
          cp1x: startX + (Math.random() - 0.5) * 200,
          cp1y: startY + (Math.random() - 0.5) * 200,
          cp2x: endX + (Math.random() - 0.5) * 200,
          cp2y: endY + (Math.random() - 0.5) * 200,
          alpha: 0.02 + Math.random() * 0.05,
          phase: Math.random() * Math.PI * 2,
          speed: 0.001 + Math.random() * 0.002,
          width: 0.5 + Math.random() * 0.8
        };
      }

      for (let i = 0; i < 18; i++) cables.push(makeCable());

      function draw() {
        ctx.clearRect(0, 0, W, H);
        const t = performance.now() / 1000;
        cables.forEach(c => {
          const a = c.alpha * (0.4 + 0.6 * Math.sin(t * c.speed * 6 + c.phase));
          if (!isFinite(a) || a < 0.003) return;
          ctx.beginPath();
          ctx.moveTo(c.x1, c.y1);
          ctx.bezierCurveTo(c.cp1x, c.cp1y, c.cp2x, c.cp2y, c.x2, c.y2);
          ctx.strokeStyle = `rgba(42,157,143,${a})`;
          ctx.lineWidth = c.width;
          ctx.stroke();
        });
        requestAnimationFrame(draw);
      }

      resize();
      window.addEventListener('resize', resize);
      draw();
    })();

    /* ══ REVEAL — teal pulse spreading ══ */
    let revealStarted = false;

    function startReveal() {
      if (revealStarted) return;
      revealStarted = true;

      const seal = document.getElementById('imageSeal');
      const container = document.getElementById('imageContainer');
      const canvas = document.getElementById('sealCanvas');
      const caption = document.getElementById('imageCaption');
      const ctx = canvas.getContext('2d');

      const W = container.offsetWidth;
      const H = container.offsetHeight;
      canvas.width = W;
      canvas.height = H;

      ctx.fillStyle = '#020808';
      ctx.fillRect(0, 0, W, H);
      seal.style.display = 'none';

      const CELL = 6;
      const cols = Math.ceil(W / CELL);
      const rows = Math.ceil(H / CELL);
      const cx = W / 2, cy = H / 2;

      const cells = [];
      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          const px = c*CELL + CELL/2, py = r*CELL + CELL/2;
          const dx = px-cx, dy = py-cy;
          const dist = Math.sqrt(dx*dx + dy*dy);
          // Cable-like spread — prefer horizontal paths
          const bias = Math.abs(dy) * 0.3;
          cells.push({ c, r, priority: dist + bias + Math.random()*10, done: false });
        }
      }
      cells.sort((a,b) => a.priority - b.priority);

      let idx = 0, frame = 0;
      const maxP = cells[cells.length-1].priority;

      function revealFrame() {
        frame++;
        const targetP = (frame / 80) * (maxP + 15);
        while (idx < cells.length && cells[idx].priority < targetP) {
          const cell = cells[idx];
          cell.done = true;
          const intensity = Math.max(0, 1 - cell.priority / maxP);
          ctx.fillStyle = `rgba(42,157,143,${0.6*intensity + 0.1})`;
          ctx.fillRect(cell.c*CELL, cell.r*CELL, CELL, CELL);
          const ci = idx;
          setTimeout(() => {
            if (cells[ci]) ctx.clearRect(cells[ci].c*CELL, cells[ci].r*CELL, CELL, CELL);
          }, 50 + Math.random() * 100);
          idx++;
        }
        if (idx < cells.length) {
          requestAnimationFrame(revealFrame);
        } else {
          setTimeout(() => {
            canvas.style.transition = 'opacity 0.5s';
            canvas.style.opacity = '0';
            setTimeout(() => { canvas.style.display = 'none'; caption.classList.add('show'); }, 500);
          }, 150);
        }
      }

      requestAnimationFrame(revealFrame);
    }

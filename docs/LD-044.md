    let crossStarted = false;

    function startCrossReveal() {
      if (crossStarted) return;
      crossStarted = true;

      const seal = document.getElementById('imageSeal');
      const container = document.getElementById('imageContainer');
      const canvas = document.getElementById('revealCanvas');
      const caption = document.getElementById('imageCaption');
      const ctx = canvas.getContext('2d');

      const W = container.offsetWidth;
      const H = container.offsetHeight;
      canvas.width = W;
      canvas.height = H;

      // Draw white seal over canvas
      ctx.fillStyle = '#fff';
      ctx.fillRect(0, 0, W, H);
      seal.style.display = 'none';

      // Cross-shaped reveal — erases from center outward in cross pattern
      // then fills the rest
      const cx = W / 2, cy = H / 2;
      let progress = 0;
      const TOTAL = 120;

      // Generate reveal cells — cross first, then fill outward
      const CELL = 8;
      const cols = Math.ceil(W / CELL);
      const rows = Math.ceil(H / CELL);
      const cells = [];

      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          const px = c * CELL + CELL/2;
          const py = r * CELL + CELL/2;
          const dx = Math.abs(px - cx);
          const dy = Math.abs(py - cy);
          // Cross proximity (arms of the cross get revealed first)
          const crossDist = Math.min(dx, dy);
          const radialDist = Math.sqrt(dx*dx + dy*dy);
          // Cells on cross arms get low priority numbers, others get higher
          const isCrossArm = dx < 40 || dy < 40;
          const priority = isCrossArm
            ? crossDist + Math.random() * 20
            : crossDist + radialDist * 0.5 + Math.random() * 40;
          cells.push({ c, r, priority, revealed: false });
        }
      }

      cells.sort((a, b) => a.priority - b.priority);
      let cellIdx = 0;
      const BATCH = Math.ceil(cells.length / TOTAL);

      function revealFrame() {
        // Reveal a batch of cells
        for (let i = 0; i < BATCH && cellIdx < cells.length; i++, cellIdx++) {
          const cell = cells[cellIdx];
          if (!cell.revealed) {
            cell.revealed = true;
            // Flash red briefly then clear
            ctx.fillStyle = `rgba(139,0,0,${0.6 - (cellIdx/cells.length)*0.4})`;
            ctx.fillRect(cell.c * CELL, cell.r * CELL, CELL, CELL);
            // Schedule clear
            setTimeout(() => {
              ctx.clearRect(cell.c * CELL, cell.r * CELL, CELL, CELL);
            }, 80 + Math.random() * 120);
          }
        }

        progress++;

        if (cellIdx < cells.length) {
          requestAnimationFrame(revealFrame);
        } else {
          // All cells revealed — fade out canvas
          setTimeout(() => {
            canvas.style.transition = 'opacity 0.5s';
            canvas.style.opacity = '0';
            setTimeout(() => {
              canvas.style.display = 'none';
              caption.classList.add('show');
            }, 500);
          }, 200);
        }
      }

      requestAnimationFrame(revealFrame);
    }
  </script>

    /* ══════════════════════════════════
       FLOATING GOLDEN EYES CANVAS
    ══════════════════════════════════ */
    (function() {
      const canvas = document.getElementById('eyeCanvas');
      const ctx = canvas.getContext('2d');
      let W, H;
      const eyes = [];

      function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
      }

      function makeEye() {
        const x = Math.random() * W;
        const y = Math.random() * H;
        const size = 4 + Math.random() * 14;
        return {
          x, y,
          size,
          phase: Math.random() * Math.PI * 2,
          speed: 0.003 + Math.random() * 0.005,
          vx: (Math.random() - 0.5) * 0.2,
          vy: (Math.random() - 0.5) * 0.1,
          blink: Math.random() * Math.PI * 2,
          blinkSpeed: 0.02 + Math.random() * 0.03,
          alpha: 0.03 + Math.random() * 0.08
        };
      }

      for (let i = 0; i < 25; i++) eyes.push(makeEye());

      function drawEye(e, t) {
        const blinkRaw = Math.sin(e.blink + t * e.blinkSpeed * 60);
        const blink = Math.max(0.05, Math.abs(blinkRaw));
        const a = e.alpha * (0.5 + 0.5 * Math.sin(e.phase + t * e.speed * 60));
        if (!isFinite(a) || a < 0.005) return;

        const rx = Math.max(1, e.size);
        const ry = Math.max(0.5, e.size * 0.45 * blink);
        const glowR = Math.max(1, rx * 1.5);

        if (!isFinite(rx) || !isFinite(ry) || !isFinite(e.x) || !isFinite(e.y)) return;

        // Eye white (very faint)
        ctx.beginPath();
        ctx.ellipse(e.x, e.y, rx, ry, 0, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(180,140,60,${a * 0.3})`;
        ctx.fill();

        // Pupil
        const pupilRx = Math.max(0.5, rx * 0.35);
        const pupilRy = Math.max(0.5, ry * 0.7);
        ctx.beginPath();
        ctx.ellipse(e.x, e.y, pupilRx, pupilRy, 0, 0, Math.PI * 2);
        ctx.fillStyle = `rgba(200,146,10,${a * 0.8})`;
        ctx.fill();

        // Glow — guard against bad values
        try {
          const g = ctx.createRadialGradient(e.x, e.y, 0, e.x, e.y, glowR);
          g.addColorStop(0, `rgba(200,146,10,${a * 0.2})`);
          g.addColorStop(1, 'rgba(200,146,10,0)');
          ctx.beginPath();
          ctx.arc(e.x, e.y, glowR, 0, Math.PI * 2);
          ctx.fillStyle = g;
          ctx.fill();
        } catch(_) {}
      }

      let last = 0;
      function frame(ts) {
        const t = ts / 1000;
        ctx.clearRect(0, 0, W, H);
        eyes.forEach(e => {
          e.x += e.vx;
          e.y += e.vy;
          if (e.x < -30) e.x = W + 30;
          if (e.x > W + 30) e.x = -30;
          if (e.y < -30) e.y = H + 30;
          if (e.y > H + 30) e.y = -30;
          drawEye(e, t);
        });
        requestAnimationFrame(frame);
      }

      resize();
      window.addEventListener('resize', resize);
      requestAnimationFrame(frame);
    })();

    /* ══════════════════════════════════
       GOLDEN EYE REVEAL — iris opening
    ══════════════════════════════════ */
    let revealStarted = false;

    function startReveal() {
      if (revealStarted) return;
      revealStarted = true;

      const seal = document.getElementById('imageSeal');
      const container = document.getElementById('imageContainer');
      const canvas = document.getElementById('sealCanvas');
      const caption = document.getElementById('imageCaption');
      const ctx = canvas.getContext('2d');

      const W = container.offsetWidth;
      const H = container.offsetHeight;
      canvas.width = W;
      canvas.height = H;

      // Draw initial cover
      ctx.fillStyle = '#040402';
      ctx.fillRect(0, 0, W, H);
      seal.style.display = 'none';

      const cx = W / 2, cy = H / 2;
      let t = 0;

      // Iris opening — golden eye radiates outward clearing cells
      const CELL = 5;
      const cols = Math.ceil(W / CELL);
      const rows = Math.ceil(H / CELL);

      const cells = [];
      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          const px = c * CELL + CELL/2;
          const py = r * CELL + CELL/2;
          const dx = px - cx, dy = py - cy;
          const dist = Math.sqrt(dx*dx + dy*dy);
          cells.push({ c, r, dist, done: false });
        }
      }
      cells.sort((a, b) => a.dist - b.dist);

      let idx = 0;
      const maxDist = Math.sqrt(cx*cx + cy*cy);
      let frame = 0;

      function revealFrame() {
        frame++;
        // Reveal in expanding circle — golden flash then clear
        const targetDist = (frame / 80) * maxDist * 1.4;
        while (idx < cells.length && cells[idx].dist < targetDist) {
          const cell = cells[idx];
          if (!cell.done) {
            cell.done = true;
            // Gold flash
            const intensity = 1 - cell.dist / maxDist;
            ctx.fillStyle = `rgba(200,146,10,${0.7 * intensity + 0.1})`;
            ctx.fillRect(cell.c * CELL, cell.r * CELL, CELL, CELL);
            const ci = idx;
            setTimeout(() => {
              ctx.clearRect(cells[ci].c * CELL, cells[ci].r * CELL, CELL, CELL);
            }, 50 + Math.random() * 80);
          }
          idx++;
        }

        if (idx < cells.length) {
          requestAnimationFrame(revealFrame);
        } else {
          setTimeout(() => {
            canvas.style.transition = 'opacity 0.5s';
            canvas.style.opacity = '0';
            setTimeout(() => {
              canvas.style.display = 'none';
              caption.classList.add('show');
            }, 500);
          }, 200);
        }
      }

      requestAnimationFrame(revealFrame);
    }

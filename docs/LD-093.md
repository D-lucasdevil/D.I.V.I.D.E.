    /* ═══ BLUE ORBS FLYING AROUND THE SCYTHE ═══ */
    (function () {
      var cv = document.getElementById('orbs');
      var ctx = cv.getContext('2d');
      var W = 0, H = 0, sc = 1;
      var dpr = Math.min(window.devicePixelRatio || 1, 1.5);
      var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      var slow = reduce ? 0.4 : 1;

      var blade = document.getElementById('bladeCenter');
      var shaft = document.getElementById('shaftCenter');

      function resize() {
        W = window.innerWidth; H = window.innerHeight;
        cv.width = Math.floor(W * dpr); cv.height = Math.floor(H * dpr);
        cv.style.width = W + 'px'; cv.style.height = H + 'px';
        ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
        sc = Math.max(0.55, Math.min(W, H) / 900);
      }
      resize();
      window.addEventListener('resize', resize);

      function sprite(inner, outer) {
        var c = document.createElement('canvas'); c.width = c.height = 64;
        var g = c.getContext('2d');
        var gr = g.createRadialGradient(32, 32, 0, 32, 32, 32);
        gr.addColorStop(0, 'rgba(255,255,255,1)');
        gr.addColorStop(0.18, inner);
        gr.addColorStop(0.5, outer);
        gr.addColorStop(1, 'rgba(20,60,160,0)');
        g.fillStyle = gr; g.fillRect(0, 0, 64, 64);
        return c;
      }
      var spr = [
        sprite('rgba(190,230,255,0.95)', 'rgba(74,163,255,0.45)'),
        sprite('rgba(140,200,255,0.95)', 'rgba(47,111,224,0.45)'),
        sprite('rgba(225,246,255,0.95)', 'rgba(120,190,255,0.40)')
      ];

      function anchor(el) {
        var r = el.getBoundingClientRect();
        return { x: r.left + r.width / 2, y: r.top + r.height / 2 };
      }

      var orbiters = [], drifters = [], i;
      for (i = 0; i < 34; i++) {
        var onBlade = Math.random() < 0.7;
        orbiters.push({
          an: onBlade ? 0 : 1,
          a: Math.random() * Math.PI * 2,
          rx: (onBlade ? 80 : 90) + Math.random() * (onBlade ? 190 : 150),
          ry: (onBlade ? 45 : 60) + Math.random() * (onBlade ? 130 : 160),
          tilt: (Math.random() - 0.5) * 1.5,
          speed: (0.22 + Math.random() * 0.75) * (Math.random() < 0.5 ? -1 : 1),
          size: 3 + Math.random() * 7,
          phase: Math.random() * 6.28,
          wob: 8 + Math.random() * 22,
          sp: (Math.random() * 3) | 0,
          trail: []
        });
      }

      function newDrifter(d, ab) {
        d.x = ab.x + (Math.random() - 0.35) * 240 * sc;
        d.y = ab.y + (Math.random() - 0.2) * 90 * sc;
        d.vx = (Math.random() - 0.5) * 26;
        d.vy = -(10 + Math.random() * 34);
        d.size = 2 + Math.random() * 5;
        d.life = 3 + Math.random() * 3.5;
        d.age = 0;
        d.phase = Math.random() * 6.28;
        d.sp = (Math.random() * 3) | 0;
        return d;
      }
      var ab0 = { x: W * 0.8, y: H * 0.2 };
      for (i = 0; i < 16; i++) {
        var d = newDrifter({}, ab0); d.age = Math.random() * d.life; drifters.push(d);
      }

      var last = performance.now();
      function frame(now) {
        var dt = Math.min(0.05, (now - last) / 1000); last = now;
        var t = now / 1000;
        var ab = anchor(blade), as = anchor(shaft);

        ctx.clearRect(0, 0, W, H);
        ctx.globalCompositeOperation = 'lighter';

        for (var k = 0; k < orbiters.length; k++) {
          var o = orbiters[k];
          var an = o.an === 0 ? ab : as;
          o.a += o.speed * dt * slow;
          var ex = o.rx * sc * Math.cos(o.a), ey = o.ry * sc * Math.sin(o.a);
          var cs = Math.cos(o.tilt), sn = Math.sin(o.tilt);
          var x = an.x + ex * cs - ey * sn + Math.sin(t * 1.3 + o.phase) * o.wob * sc;
          var y = an.y + ex * sn + ey * cs + Math.cos(t * 1.1 + o.phase) * o.wob * sc;
          o.trail.unshift({ x: x, y: y });
          if (o.trail.length > 10) o.trail.pop();
          var pulse = 0.85 + 0.15 * Math.sin(t * 2 + o.phase);
          for (var j = o.trail.length - 1; j >= 0; j--) {
            var p = o.trail[j];
            var f = 1 - j / o.trail.length;
            var s = o.size * 7 * sc * pulse * (0.35 + 0.65 * f);
            ctx.globalAlpha = (j === 0 ? 0.95 : 0.32 * f);
            ctx.drawImage(spr[o.sp], p.x - s / 2, p.y - s / 2, s, s);
          }
        }

        for (var m = 0; m < drifters.length; m++) {
          var q = drifters[m];
          q.age += dt * slow;
          if (q.age >= q.life) { newDrifter(q, ab); continue; }
          q.x += (q.vx + Math.sin(t * 1.5 + q.phase) * 12) * dt * slow;
          q.y += q.vy * dt * slow;
          var u = q.age / q.life;
          var al = u < 0.2 ? u / 0.2 : 1 - (u - 0.2) / 0.8;
          var sz = q.size * 8 * sc * (0.7 + 0.5 * Math.sin(t * 3 + q.phase) * 0.3 + 0.3);
          ctx.globalAlpha = Math.max(0, Math.min(1, al * 0.85));
          ctx.drawImage(spr[q.sp], q.x - sz / 2, q.y - sz / 2, sz, sz);
        }

        ctx.globalAlpha = 1;
        ctx.globalCompositeOperation = 'source-over';
        requestAnimationFrame(frame);
      }
      requestAnimationFrame(frame);
    })();
  </script>

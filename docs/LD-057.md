    /* Butterfly canvas — hand-drawn wing shapes floating */
    (function() {
      const canvas = document.getElementById('butterflyCanvas');
      const ctx = canvas.getContext('2d');
      let W, H;

      const hues = [280, 200, 160, 30, 320, 60, 180];
      const butterflies = [];

      function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
      }

      function makeBf() {
        return {
          x: Math.random() * W,
          y: Math.random() * H,
          vx: (Math.random() - 0.5) * 0.6,
          vy: (Math.random() - 0.5) * 0.4,
          waveX: Math.random() * Math.PI * 2,
          waveY: Math.random() * Math.PI * 2,
          waveSpeedX: 0.008 + Math.random() * 0.012,
          waveSpeedY: 0.005 + Math.random() * 0.008,
          waveAmpX: 30 + Math.random() * 50,
          waveAmpY: 20 + Math.random() * 30,
          size: 6 + Math.random() * 14,
          hue: hues[Math.floor(Math.random() * hues.length)],
          alpha: 0.04 + Math.random() * 0.10,
          flapPhase: Math.random() * Math.PI * 2,
          flapSpeed: 0.08 + Math.random() * 0.12,
          angle: (Math.random() - 0.5) * 0.4
        };
      }

      for (let i = 0; i < 20; i++) butterflies.push(makeBf());

      function drawBf(b, t) {
        const flap = Math.sin(b.flapPhase + t * b.flapSpeed * 60);
        const scaleX = Math.abs(flap) * 0.9 + 0.1;
        const a = b.alpha * (0.5 + 0.5 * Math.abs(Math.sin(t * 0.3 + b.wavePhase || 0)));
        if (!isFinite(a) || a < 0.005) return;

        ctx.save();
        ctx.translate(b.x, b.y);
        ctx.rotate(b.angle);

        const s = b.size;

        // Left wings
        ctx.save();
        ctx.scale(-scaleX, 1);
        ctx.beginPath();
        ctx.ellipse(-s*0.6, -s*0.4, s*0.8, s*0.5, 0.3, 0, Math.PI * 2);
        ctx.fillStyle = `hsla(${b.hue},70%,65%,${a * 0.8})`;
        ctx.fill();
        ctx.beginPath();
        ctx.ellipse(-s*0.5, s*0.3, s*0.6, s*0.35, -0.2, 0, Math.PI * 2);
        ctx.fillStyle = `hsla(${b.hue + 30},60%,60%,${a * 0.6})`;
        ctx.fill();
        ctx.restore();

        // Right wings
        ctx.save();
        ctx.scale(scaleX, 1);
        ctx.beginPath();
        ctx.ellipse(s*0.6, -s*0.4, s*0.8, s*0.5, -0.3, 0, Math.PI * 2);
        ctx.fillStyle = `hsla(${b.hue},70%,65%,${a * 0.8})`;
        ctx.fill();
        ctx.beginPath();
        ctx.ellipse(s*0.5, s*0.3, s*0.6, s*0.35, 0.2, 0, Math.PI * 2);
        ctx.fillStyle = `hsla(${b.hue + 30},60%,60%,${a * 0.6})`;
        ctx.fill();
        ctx.restore();

        ctx.restore();
      }

      function draw() {
        ctx.clearRect(0, 0, W, H);
        const t = performance.now() / 1000;
        butterflies.forEach(b => {
          b.waveX += b.waveSpeedX;
          b.waveY += b.waveSpeedY;
          b.x += b.vx + Math.sin(b.waveX) * 0.5;
          b.y += b.vy + Math.cos(b.waveY) * 0.3;
          if (b.x < -60) b.x = W + 60;
          if (b.x > W + 60) b.x = -60;
          if (b.y < -60) b.y = H + 60;
          if (b.y > H + 60) b.y = -60;
          try { drawBf(b, t); } catch(_) {}
        });
        requestAnimationFrame(draw);
      }

      resize();
      window.addEventListener('resize', resize);
      draw();
    })();
  </script>

    /* ══ FROST CANVAS — icy drift blobs ══ */
    (function() {
      const canvas = document.getElementById('frostCanvas');
      const ctx = canvas.getContext('2d');
      let W, H;
      const blobs = [];

      function resize() {
        W = window.innerWidth;
        H = window.innerHeight;
        canvas.width = W;
        canvas.height = H;
      }

      for (let i = 0; i < 20; i++) {
        blobs.push({
          x: Math.random() * (window.innerWidth || 800),
          y: Math.random() * (window.innerHeight || 600),
          r: 60 + Math.random() * 160,
          vx: (Math.random() - 0.5) * 0.25,
          vy: (Math.random() - 0.5) * 0.12,
          alpha: 0.015 + Math.random() * 0.04,
          phase: Math.random() * Math.PI * 2,
          speed: 0.002 + Math.random() * 0.004
        });
      }

      function draw() {
        ctx.clearRect(0, 0, W, H);
        const t = performance.now() / 1000;
        blobs.forEach(b => {
          b.x += b.vx;
          b.y += b.vy;
          if (b.x < -b.r*2) b.x = W + b.r;
          if (b.x > W+b.r*2) b.x = -b.r;
          if (b.y < -b.r*2) b.y = H + b.r;
          if (b.y > H+b.r*2) b.y = -b.r;
          const a = b.alpha * (0.5 + 0.5*Math.sin(t*b.speed*6+b.phase));
          if (!isFinite(a) || a < 0.001 || !isFinite(b.x) || !isFinite(b.y) || b.r < 1) return;
          try {
            const g = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.r);
            g.addColorStop(0, `rgba(168,216,240,${a})`);
            g.addColorStop(0.5, `rgba(140,190,220,${a*0.5})`);
            g.addColorStop(1, 'rgba(100,160,200,0)');
            ctx.fillStyle = g;
            ctx.beginPath();
            ctx.arc(b.x, b.y, b.r, 0, Math.PI*2);
            ctx.fill();
          } catch(_) {}
        });
        requestAnimationFrame(draw);
      }

      resize();
      window.addEventListener('resize', resize);
      draw();
    })();

    /* ══ ICE SHARD REVEAL ══ */
    let revealStarted = false;

    function startReveal() {
      if (revealStarted) return;
      revealStarted = true;

      const seal = document.getElementById('imageSeal');
      const container = document.getElementById('imageContainer');
      const canvas = document.getElementById('sealCanvas');
      const caption = document.getElementById('imageCaption');
      const ctx = canvas.getContext('2d');

      const W = container.offsetWidth;
      const H = container.offsetHeight;
      canvas.width = W;
      canvas.height = H;

      ctx.fillStyle = '#020608';
      ctx.fillRect(0, 0, W, H);
      seal.style.display = 'none';

      // Shatter from center — triangular ice shards
      // Represent as a grid of cells that crack away
      const CELL = 7;
      const cols = Math.ceil(W / CELL);
      const rows = Math.ceil(H / CELL);
      const cx = W/2, cy = H/2;

      const cells = [];
      for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
          const px = c*CELL + CELL/2, py = r*CELL + CELL/2;
          const dx = px-cx, dy = py-cy;
          const dist = Math.sqrt(dx*dx + dy*dy);
          // Shatter pattern — radial with crack-like variation
          const angle = Math.atan2(dy, dx);
          const variation = Math.sin(angle * 8) * 20;
          cells.push({ c, r, priority: dist + variation + Math.random()*15, done: false });
        }
      }
      cells.sort((a,b) => a.priority - b.priority);

      let idx = 0;
      let frame = 0;
      const maxDist = Math.sqrt(cx*cx + cy*cy);

      function revealFrame() {
        frame++;
        const targetPriority = (frame / 70) * (maxDist + 40);

        while (idx < cells.length && cells[idx].priority < targetPriority) {
          const cell = cells[idx];
          cell.done = true;
          // Ice blue flash
          const intensity = Math.max(0, 1 - cell.priority / maxDist);
          ctx.fillStyle = `rgba(168,216,240,${0.6*intensity + 0.15})`;
          ctx.fillRect(cell.c*CELL, cell.r*CELL, CELL, CELL);
          const ci = idx;
          setTimeout(() => {
            if (cells[ci]) ctx.clearRect(cells[ci].c*CELL, cells[ci].r*CELL, CELL, CELL);
          }, 40 + Math.random()*100);
          idx++;
        }

        if (idx < cells.length) {
          requestAnimationFrame(revealFrame);
        } else {
          setTimeout(() => {
            canvas.style.transition = 'opacity 0.5s';
            canvas.style.opacity = '0';
            setTimeout(() => { canvas.style.display = 'none'; caption.classList.add('show'); }, 500);
          }, 150);
        }
      }

      requestAnimationFrame(revealFrame);
    }
